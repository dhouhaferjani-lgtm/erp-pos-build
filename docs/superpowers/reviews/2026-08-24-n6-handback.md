```sql
SELECT company_id, type, chain_sequence, COUNT(*) AS forks
  FROM documents
 WHERE fiscal_hash IS NOT NULL
 GROUP BY company_id, type, chain_sequence
HAVING COUNT(*) > 1
 ORDER BY company_id, type, chain_sequence;

-- (b) a non-genesis document with no predecessor recorded
SELECT id, document_number, type, status, chain_sequence
  FROM documents
 WHERE fiscal_hash IS NOT NULL AND previous_hash IS NULL AND chain_sequence > 1;

-- (c) a broken link: previous_hash naming no sealed document of the same chain
SELECT d.id, d.document_number, d.type, d.chain_sequence
  FROM documents d
 WHERE d.fiscal_hash IS NOT NULL AND d.previous_hash IS NOT NULL
   AND NOT EXISTS (SELECT 1 FROM documents p
                    WHERE p.company_id = d.company_id AND p.type = d.type
                      AND p.fiscal_hash = d.previous_hash);
```
